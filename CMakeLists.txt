cmake_minimum_required(VERSION 3.21)
option(DATAPUMP_BUILD_GUI "Build the native C++ desktop application" ON)
option(DATAPUMP_PORTABLE "Install a relocatable bundle with native dependencies" ON)
option(DATAPUMP_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(DATAPUMP_PORTABLE)
  set(OPENSSL_USE_STATIC_LIBS TRUE CACHE BOOL "Link OpenSSL statically")
  set(OPENSSL_MSVC_STATIC_RT TRUE CACHE BOOL "Use OpenSSL built for the static MSVC runtime")
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "MSVC runtime")
endif()
project(DataPump VERSION 0.5.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(datapump src/crypto.cpp src/packet.cpp src/modem.cpp src/streaming_modem.cpp src/channel.cpp src/runtime.cpp src/audio.cpp src/resampler.cpp src/qr.cpp src/transfer.cpp src/tuning.cpp src/live.cpp src/signal_view.cpp third_party/qrcodegen/qrcodegen.cpp)
target_include_directories(datapump PUBLIC include)
target_include_directories(datapump PRIVATE third_party/qrcodegen)
target_link_libraries(datapump PUBLIC OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
if(DATAPUMP_PORTABLE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND NOT DATAPUMP_SANITIZERS)
  target_link_options(datapump PUBLIC -static-libstdc++ -static-libgcc)
endif()
if(WIN32)
  target_link_libraries(datapump PRIVATE winmm advapi32)
endif()
if(MSVC)
  target_compile_options(datapump PRIVATE /W4 /permissive-)
else()
  target_compile_options(datapump PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(DATAPUMP_SANITIZERS)
    target_compile_options(datapump PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(datapump PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(NOT MSVC)
  # Preserve the pinned upstream QR source; project-only conversion warnings
  # do not imply a requirement to modify vendored, independently tested code.
  set_source_files_properties(third_party/qrcodegen/qrcodegen.cpp PROPERTIES COMPILE_OPTIONS "-Wno-conversion")
endif()
add_executable(pump src/main.cpp)
target_link_libraries(pump PRIVATE datapump)
target_compile_definitions(pump PRIVATE DATAPUMP_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_link_options(pump PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/pump.manifest")
endif()
set(datapump_executables pump)
if(DATAPUMP_BUILD_GUI)
  include(cmake/NativeGui.cmake)
  list(APPEND datapump_executables datapump-gui)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(benchmark_receiver EXCLUDE_FROM_ALL tools/benchmark_receiver.cpp)
  target_link_libraries(benchmark_receiver PRIVATE datapump)
  foreach(component crypto keyring packet modem streaming_modem regressions runtime qr transfer tuning live resampler audio_rates signal_view channel)
    add_executable(test_${component} tests/test_${component}.cpp)
    target_link_libraries(test_${component} PRIVATE datapump)
    add_test(NAME ${component} COMMAND test_${component})
  endforeach()
  if(DATAPUMP_BUILD_GUI)
    add_executable(test_gui_plots tests/test_gui_plots.cpp)
    target_link_libraries(test_gui_plots PRIVATE datapump)
    add_test(NAME gui_plots COMMAND test_gui_plots)
    add_executable(test_gui_native tests/test_gui_native.cpp src/gui/state.cpp)
    target_include_directories(test_gui_native PRIVATE src/gui)
    target_link_libraries(test_gui_native PRIVATE datapump)
    add_test(NAME gui_native_policy COMMAND test_gui_native)
    add_test(NAME gui_self_check COMMAND datapump-gui --self-check)
  endif()
  if(UNIX)
    add_library(alsa_stub SHARED tests/alsa_stub.cpp)
    set_target_properties(alsa_stub PROPERTIES OUTPUT_NAME asound SOVERSION 2
      LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/audio-test-lib")
    add_executable(test_alsa_audio tests/test_alsa_audio.cpp src/audio.cpp src/resampler.cpp)
    target_include_directories(test_alsa_audio PRIVATE include tests)
    target_link_libraries(test_alsa_audio PRIVATE alsa_stub Threads::Threads ${CMAKE_DL_LIBS})
    add_test(NAME alsa_audio_contract COMMAND test_alsa_audio)
    add_executable(test_windows_audio tests/test_windows_audio.cpp src/audio.cpp src/resampler.cpp)
    target_include_directories(test_windows_audio PRIVATE tests/winmm_stub include)
    target_compile_definitions(test_windows_audio PRIVATE _WIN32)
    if(DATAPUMP_SANITIZERS AND NOT MSVC)
      foreach(audio_test test_windows_audio test_alsa_audio alsa_stub)
        target_compile_options(${audio_test} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${audio_test} PRIVATE -fsanitize=address,undefined)
      endforeach()
    endif()
    add_test(NAME windows_audio_contract COMMAND test_windows_audio)
  endif()
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:pump>)
  endif()
endif()
install(FILES README.md LICENSE DESTINATION share/doc/datapump)
install(DIRECTORY docs/ DESTINATION share/doc/datapump/docs)
install(FILES third_party/qrcodegen/LICENSE DESTINATION share/doc/datapump RENAME QR-LICENSE)
if(DATAPUMP_BUILD_GUI)
  install(FILES third_party/fltk/COPYING DESTINATION share/doc/datapump RENAME FLTK-LICENSE)
endif()
include(cmake/Portable.cmake)
datapump_install_native(TARGETS ${datapump_executables})
if(BUILD_TESTING AND DATAPUMP_PORTABLE)
  add_test(NAME packaging_support COMMAND ${CMAKE_COMMAND}
    -DBINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}/packaging-fixture
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/packaging_support.cmake)
  if(NOT DATAPUMP_SANITIZERS)
    add_test(NAME native_relocation COMMAND ${CMAKE_COMMAND}
      -DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR} -DCONFIG=$<CONFIG>
      -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/package_native.cmake)
  endif()
endif()
set(CPACK_PACKAGE_NAME DataPump)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "DataPump-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}-native")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/releases")
set(CPACK_GENERATOR "TGZ;ZIP")
include(CPack)
