cmake_minimum_required(VERSION 3.21)
project(DataPumpRevEventProbe LANGUAGES CXX)
get_filename_component(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
enable_testing()
add_executable(test_rev_event_pump "${SOURCE_ROOT}/tests/test_rev_event_pump.cpp")
set_target_properties(test_rev_event_pump PROPERTIES
  CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO
  MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(test_rev_event_pump PRIVATE "${SOURCE_ROOT}/third_party/rev/external")
if(WIN32)
  target_link_libraries(test_rev_event_pump PRIVATE user32)
else()
  target_include_directories(test_rev_event_pump PRIVATE "${SOURCE_ROOT}/tests/rev_event_stub")
endif()
if(MSVC)
  target_compile_options(test_rev_event_pump PRIVATE /utf-8 /permissive-)
endif()
add_test(NAME rev_event_pump COMMAND test_rev_event_pump)
set_tests_properties(rev_event_pump PROPERTIES TIMEOUT 10)
